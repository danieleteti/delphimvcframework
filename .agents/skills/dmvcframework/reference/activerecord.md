Entity mapping, CRUD, queries, RQL, hooks, soft delete, audit columns, optimistic locking, master-detail.

## Entity / ActiveRecord Model Scaffolding

**File: `Entities.MyResource.pas`**

```delphi
unit Entities.MyResource;

interface

uses
  MVCFramework.ActiveRecord,
  MVCFramework.Serializer.Commons,
  MVCFramework.Nullables,
  System.Generics.Collections;

type
  [MVCNameCase(ncCamelCase)]
  [MVCTable('my_resources')]
  TMyResource = class(TMVCActiveRecord)
  private
    [MVCTableField('id', [foPrimaryKey, foAutoGenerated])]
    fID: NullableInt64;

    [MVCTableField('name')]
    fName: String;

    [MVCTableField('description')]
    fDescription: String;

    [MVCTableField('created_at', [foReadOnly])]
    fCreatedAt: TDateTime;

    [MVCTableField('is_active')]
    fIsActive: Boolean;
  public
    property ID: NullableInt64 read fID write fID;
    property Name: String read fName write fName;
    property Description: String read fDescription write fDescription;
    property CreatedAt: TDateTime read fCreatedAt;
    property IsActive: Boolean read fIsActive write fIsActive;
  end;

implementation

end.
```

### ActiveRecord Field Options

| Option | Meaning |
|--------|---------|
| `foPrimaryKey` | Primary key column |
| `foAutoGenerated` | DB generates the value (IDENTITY / SERIAL / AUTOINCREMENT) |
| `foReadOnly` | Never written by the ORM (computed, audit columns) |
| `foVersion` | Optimistic locking field |
| `foDoNotSelect` | Excluded from SELECT |
| `foDoNotInsert` | Excluded from INSERT |
| `foDoNotUpdate` | Excluded from UPDATE |
| `foRefresh` | Re-read from DB after INSERT/UPDATE |

### Serialization Attributes

```delphi
[MVCNameCase(ncCamelCase)]       // properties → camelCase in JSON
```

### ActiveRecord Operations

```delphi
// Insert
entity.Insert;

// Update
entity.Update;

// Insert or update (upsert)
entity.Store;

// Delete
entity.Delete;

// Get by PK (raises exception if not found)
entity := TMVCActiveRecord.GetByPk<TMyResource>(id);

// Get all records
list := TMVCActiveRecord.All<TMyResource>;

// Filter with WHERE clause
list := TMVCActiveRecord.Where<TMyResource>('name = ?', ['foo']);

// RQL query (Resource Query Language)
list := TMVCActiveRecord.SelectRQL<TMyResource>('eq(name,"foo")', 100);

// Raw SQL dataset
ds := TMVCActiveRecord.SelectDataSet('SELECT * FROM my_resources WHERE id=?', [id], True);

// Count
count := TMVCActiveRecord.Count<TMyResource>;

// Transaction
TMVCActiveRecord.CurrentConnection.StartTransaction;
try
  // ... operations
  TMVCActiveRecord.CurrentConnection.Commit;
except
  TMVCActiveRecord.CurrentConnection.Rollback;
  raise;
end;

// Merge (diff + apply for master-detail updates)
TMVCActiveRecord.Merge<TDetail>(existingList, incomingList).Apply(
  procedure(const Item: TDetail; const Action: TMVCEntityAction; var Handled: Boolean)
  begin
    case Action of
      eaCreate: begin { add to parent } Handled := True; end;
      eaUpdate: begin { update in parent } Handled := True; end;
      eaDelete: begin { remove from parent } Handled := False; end; // False = auto-delete
    end;
  end);
```

### Named SQL and RQL queries

Declare reusable queries as class-level attributes; optionally pin them to a specific backend:

```delphi
[MVCTable('customers')]
[MVCNamedSQLQuery('BestCustomers', 'SELECT * FROM customers WHERE rating >= 4')]
[MVCNamedSQLQuery('WithRating',    'SELECT * FROM customers WHERE rating = ?')]
[MVCNamedSQLQuery('PGProc', 'SELECT * FROM sp_get_customers()', TMVCActiveRecordBackEnd.PostgreSQL)]
[MVCNamedRQLQuery('HighRating', 'ge(rating,4);sort(+code)')]
[MVCNamedRQLQuery('RatingEq',   'eq(rating,%d);sort(+code)')]
TCustomer = class(TMVCActiveRecord)
  ...
end;

// Execute named SQL query (params are positional ?)
var lList := TMVCActiveRecord.SelectByNamedQuery<TCustomer>('BestCustomers', [], []);
var lList := TMVCActiveRecord.SelectByNamedQuery<TCustomer>('WithRating', [3], [ftInteger]);

// Ignore missing fields (useful for stored procedures / views)
var lList := TMVCActiveRecord.SelectByNamedQuery<TCustomer>(
  'PGProc', [], [], [loIgnoreNotExistentFields]);

// Execute named RQL query (RQL text is used as a format string with the params)
var lList := TMVCActiveRecord.SelectRQLByNamedQuery<TCustomer>('HighRating', [], 1000);
var lList := TMVCActiveRecord.SelectRQLByNamedQuery<TCustomer>('RatingEq', [3], 1000);

// Check at runtime whether a named query exists
var lSQL: TSQLQueryWithName;
if TMVCActiveRecord.TryGetSQLQuery<TCustomer>('BestCustomers', lSQL) then
  WriteLn(lSQL.Name, ' → ', lSQL.SQL);

var lRQL: TRQLQueryWithName;
if TMVCActiveRecord.TryGetRQLQuery<TCustomer>('HighRating', lRQL) then
  WriteLn(lRQL.Name, ' → ', lRQL.RQLText);
```

### Additional query methods

```delphi
// GetByPk(id, False) → returns nil instead of raising when not found
var lCust := TMVCActiveRecord.GetByPK<TCustomer>(id, False);
if lCust = nil then { not found };

// First record matching a SQL WHERE clause (nil if none)
var lCust := TMVCActiveRecord.GetFirstByWhere<TCustomer>('id > ?', [1]);
var lCust := TMVCActiveRecord.GetFirstByWhere<TCustomer>('id > ?', [1], [ftInteger]);
// Raise := False → return nil instead of raising when nothing matches
var lCust := TMVCActiveRecord.GetFirstByWhere<TCustomer>('rating = 5', [], False);

// Exactly one record (raises if none or more than one)
var lCust := TMVCActiveRecord.GetOneByWhere<TCustomer>('code = ?', ['ABC']);
var lCust := TMVCActiveRecord.GetOneByWhere<TCustomer>('code = ?', ['ABC'], [ftString]);

// Exactly one record by RQL (True = raise if > 1; pass False to return nil if none)
var lCust := TMVCActiveRecord.SelectOneByRQL<TCustomer>('eq(code,"ABC")', True);

// Delete all records of a class
TMVCActiveRecord.DeleteAll(TCustomer);    // class reference form
TMVCActiveRecord.DeleteAll<TCustomer>;    // generic form

// Delete by RQL filter
TMVCActiveRecord.DeleteRQL(TCustomer, 'lt(rating,3)');

// Count with optional RQL filter
var n := TMVCActiveRecord.Count<TCustomer>;
var n := TMVCActiveRecord.Count<TCustomer>('eq(city,"Rome")');
var n := TMVCActiveRecord.Count(TCustomer, 'eq(city,"Rome")');  // class reference form

// SelectRQL with -1 = no limit
var lAll := TMVCActiveRecord.SelectRQL<TCustomer>('sort(+code)', -1);

// Load into an existing instance (no new allocation)
lCust.LoadByPK(id);  // returns False if not found

// Reload from DB (useful after DB triggers rewrite computed fields)
lCust.Refresh;

// Dynamic property access by mapped property name
lCust.Attributes['CompanyName'] := 'Acme Corp.';
var v := lCust.Attributes['Rating'].AsInteger;

// Primary key helpers
if lCust.PKIsNull then { new, unsaved entity };
lCust.SetPK(TValue.From<NullableString>(newCode));  // useful in OnBeforeInsert for client-generated PKs
```

### Change tracking

`[MVCChangeTracking]` on the class makes `Update` emit only changed columns (CRC32 snapshot per field). Use `UpdateIfChanged` to skip the SQL round-trip entirely when nothing changed:

```delphi
[MVCTable('customers')]
[MVCChangeTracking]
TTrackedCustomer = class(TMVCActiveRecord)
  ...
end;

// Only the changed column is sent: UPDATE customers SET city=? WHERE id=?
var lCust := TMVCActiveRecord.GetByPK<TTrackedCustomer>(id);
try
  lCust.City := 'Paris';
  lCust.Update;
finally
  lCust.Free;
end;

// Inspect change state
if lCust.IsChanged then ...           // any field changed?
if lCust.IsChanged('City') then ...   // specific field (case-insensitive)
var fields := lCust.GetChangedFields; // TArray<string>

// No-op when nothing changed; equivalent to Update when dirty
lCust.UpdateIfChanged;
```

### Soft delete

Put `[MVCSoftDeleted]` on a `NullableTDateTime` field. `Delete` sets the marker to `Now()`; every `SELECT` auto-adds `WHERE deleted_at IS NULL` unless overridden:

```delphi
[MVCTable('notes')]
TNoteEntity = class(TMVCActiveRecord)
private
  [MVCTableField('id', [foPrimaryKey, foAutoGenerated])]
  fID: Integer;
  [MVCTableField('description')]
  fDescription: NullableString;
  [MVCTableField('deleted_at')]
  [MVCSoftDeleted]
  fDeletedAt: NullableTDateTime;  // MUST be NullableTDateTime, not TDateTime
public
  property ID: Integer read fID write fID;
  property Description: NullableString read fDescription write fDescription;
  property DeletedAt: NullableTDateTime read fDeletedAt write fDeletedAt;
end;

note.Delete;   // sets deleted_at = Now()
note.Restore;  // clears deleted_at (undelete)
note.HardDelete; // physical DELETE regardless of marker

// Bulk operations via RQL
TMVCActiveRecord.HardDeleteRQL<TNoteEntity>('lt(id,100)');
TMVCActiveRecord.RestoreRQL<TNoteEntity>('eq(id,42)');

// Include soft-deleted rows in the next SELECT (thread-local toggle)
TMVCActiveRecord.IncludeSoftDeleted(True);
var allIncludingDeleted := TMVCActiveRecord.All<TNoteEntity>;
TMVCActiveRecord.IncludeSoftDeleted(False);  // reset to default
```

### Optimistic locking (foVersion)

```delphi
[MVCTable('orders')]
TVersionedOrder = class(TMVCActiveRecord)
private
  [MVCTableField('id', [foPrimaryKey, foAutoGenerated])]
  fID: NullableInt64;
  [MVCTableField('objversion', [foVersion])]
  fObjVersion: Integer;
public
  // read/write: the client must send back the version it read, so the round-trip works
  property ObjVersion: Integer read fObjVersion write fObjVersion;
end;

// UPDATE / DELETE emit: WHERE id = ? AND objversion = <the value that was loaded>
// No row matched (someone else changed it meanwhile) →
//   EMVCActiveRecordVersionedItemNotFound, which descends from EMVCActiveRecordNotFound
//   and therefore surfaces as HTTP 404 — NOT 409. If you want a 409 for the client,
//   catch it and re-raise.
```

`Update` and `Delete` both take `RaiseExceptionIfNotFound: Boolean = True`:

```delphi
procedure Update(const RaiseExceptionIfNotFound: Boolean = True);
procedure Delete(const RaiseExceptionIfNotFound: Boolean = True);
```

### UseTransactionContext — automatic commit/rollback

```delphi
// Scoped guard: commits on normal scope exit, rolls back if an exception escapes
begin
  var Ctx := TMVCActiveRecord.UseTransactionContext;

  var lCustomer := TCustomer.Create;
  try
    lCustomer.CompanyName := 'Transaction Inc.';
    lCustomer.Insert;

    var lOrder := TOrder.Create;
    try
      lOrder.CustomerID := lCustomer.ID;
      lOrder.Insert;
    finally
      lOrder.Free;
    end;
  finally
    lCustomer.Free;
  end;
  // Ctx goes out of scope → auto-commit (or rollback on exception)
end;
```

## Master-Detail Pattern

```delphi
[MVCTable('orders')]
TOrder = class(TMVCActiveRecord)
private
  [MVCTableField('id', [foPrimaryKey, foAutoGenerated])]
  fID: NullableInt64;
  [MVCOwned]                      // framework frees this list automatically
  fDetails: TObjectList<TOrderLine>;
protected
  procedure OnAfterLoad; override;
  procedure OnAfterInsertOrUpdate; override;
public
  property ID: NullableInt64 read fID write fID;
  property Lines: TObjectList<TOrderLine> read fDetails;
end;

procedure TOrder.OnAfterLoad;
var
  lList: TObjectList<TOrderLine>;
begin
  inherited;
  lList := TMVCActiveRecord.SelectRQL<TOrderLine>(
    Format('eq(orderId,%d)', [ID.Value]), 1000);
  try
    fDetails.Clear;
    fDetails.AddRange(lList);
    lList.OwnsObjects := False;
  finally
    lList.Free;
  end;
end;

procedure TOrder.OnAfterInsertOrUpdate;
begin
  inherited;
  for var lLine in fDetails do
  begin
    lLine.OrderID := ID.Value;
    lLine.Store;
  end;
end;
```

## Connection per request — `TMVCActiveRecordMiddleware`

ActiveRecord needs a connection bound to the current thread. In a server, that is the middleware's job:
it opens and binds the connection before routing and releases it after the action. Do **not** call
`ActiveRecordConnectionsRegistry.AddDefaultConnection` from controller code.

```delphi
// the connection def is already registered (e.g. by SetupDatabaseConnection at startup)
AEngine.AddMiddleware(TMVCActiveRecordMiddleware.Create('activerecorddb'));

// ...or let the middleware load the def from an ini file
AEngine.AddMiddleware(TMVCActiveRecordMiddleware.Create('activerecorddb', 'FDConnectionDefs.ini'));

// ...or bind extra named connections too (several databases in one request)
AEngine.AddMiddleware(TMVCActiveRecordMiddleware.Create('maindb',
  ['reporting'], ['reportingdb'], 'FDConnectionDefs.ini'));
```

Required whenever a controller touches `TMVCActiveRecord`, and **mandatory** for
`TMVCActiveRecordController` (below).

## RQL straight from the client

Let the caller filter, and keep a bound on the row count:

```delphi
[MVCPath] [MVCHTTPMethods([httpGET])]
function GetCustomers([MVCFromQueryString('rql', '')] RQLFilter: String): TObjectList<TCustomer>;
begin
  Result := TMVCActiveRecord.SelectRQL<TCustomer>(RQLFilter, 1000);   // empty RQL → everything
end;
```

RQL is parsed, not concatenated, so it cannot smuggle SQL — but it lets the caller filter on **any mapped
column**. Only expose it where every column is safe to filter on, and always combine it with the caller's
ownership scope (see the `dmvcframework-security` skill).

## Automatic REST CRUD — `TMVCActiveRecordController`

A whole CRUD API over your entities, with no controller of your own. Register the controller, the
middleware, and the entities you want exposed:

```delphi
// EngineConfigU.pas
AEngine.AddController(TMVCActiveRecordController, '/api/entities');
AEngine.AddMiddleware(TMVCActiveRecordMiddleware.Create(CON_DEF_NAME));   // REQUIRED

// EntitiesU.pas — the URL segment for each entity
initialization
  ActiveRecordMappingRegistry.AddEntity('people', TPerson);
  ActiveRecordMappingRegistry.AddEntity('articles', TArticle);
```

The URL scheme you get:

| Verb + path | What it does |
|-------------|--------------|
| `GET /api/entities/(entity)?rql=…&count=true` | list, RQL-filtered; `count=true` adds `meta.count` |
| `GET /api/entities/(entity)/searches?rql=…` | same as the list |
| `POST /api/entities/(entity)/searches` body `{"rql":"…"}` | list via POST, for long filters |
| `GET /api/entities/(entity)/(id)` | one, 404 if absent |
| `POST /api/entities/(entity)?refresh=true` | create → 201 + `X-REF` header |
| `PUT /api/entities/(entity)/(id)?refresh=true` | update |
| `DELETE /api/entities/(entity)/(id)` | delete — idempotent, never 404 |

Page size comes from `Config[TMVCConfigKey.MaxEntitiesRecordCount]` (default 20).

**This is wide open by default.** Gate it with the authorization callback, and restrict per entity with
`[MVCEntityActions(...)]`:

```delphi
AEngine.AddController(TMVCActiveRecordController,
  function: TMVCController
  begin
    Result := TMVCActiveRecordController.Create(
      function(Context: TWebContext; aClass: TMVCActiveRecordClass;
               aAction: TMVCActiveRecordAction): Boolean
      begin
        Result := (aAction = TMVCActiveRecordAction.Retrieve) or
                  Context.LoggedUser.Roles.Contains('admin');
      end, '/api/entities');
  end, '/api/entities');
```

### Overriding one verb — `IMVCEntityProcessor`

When a single entity needs custom behaviour, do not abandon the auto-CRUD; intercept it.
`Handled := False` falls back to the default.

```delphi
  const entityname: string; var Handled: Boolean);

  Handled := True;
  Renderer.Render(ObjectDict().Add('data', TMVCActiveRecord.All<TArticle>));
end;

initialization
```

## Bulk operations

The transaction-context guard is the idiom — one transaction, no `try/except`:

```delphi
function TCustomersController.BulkCreateCustomers(
  [MVCFromBody] const Customers: TObjectList<TCustomer>): IMVCResponse;
begin
  begin
    var lCtx := TMVCActiveRecord.UseTransactionContext;   // commits at scope exit, rolls back on exception
    for var lCustomer in Customers do
      lCustomer.Insert;
  end;
  Result := CreatedResponse();
end;
```

**Reusing one instance in a loop:** clear the nullable PK before each `Insert`, or the entity looks like an
existing row.

```delphi
for var I := 1 to 100 do
begin
  lCustomer.ID.Clear;              // otherwise Store/Insert would target the previous row
  lCustomer.CompanyName := 'Company ' + I.ToString;
  lCustomer.Insert;
end;
```

### `Merge` — synchronize a list against the database

```delphi
class function Merge<T: TMVCActiveRecord>(CurrentList, NewList: TObjectList<T>;
  const MergeMode: TMergeMode = [mmInsert, mmUpdate, mmDelete]): IMVCMultiExecutor<T>;
```

Rows in `NewList` **with** a PK are updated, **without** a PK are inserted, and rows only in `CurrentList`
are deleted. Restrict the modes to opt out of any of the three.

```delphi
TMVCActiveRecord.Merge<TCustomer>(lCurrentList, lNewList).Apply(
  procedure(const Customer: TCustomer; const EntityAction: TMVCEntityAction; var Handled: Boolean)
  begin
    Handled := False;   // False = let the framework perform the insert/update/delete
  end);
```
